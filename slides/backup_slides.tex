\begin{frame}{Geometry: why the beam count is random}
  \begin{columns}[c]
    \column{0.5\textwidth}
    \small
    Beams at $x_j = u + j\,s$, \ $s = R\,\Delta\theta$, \ offset $u \sim U[0,s)$\\[0.5em]
    Object side $[0, W]$, \ $L = W/s$:
    \[
      n_h = \begin{cases}
        \lfloor L\rfloor + 1 & \text{if } u < W - \lfloor L\rfloor s \\
        \lfloor L\rfloor & \text{otherwise}
      \end{cases}
    \]
    \[
      \Rightarrow\ P(n_h = \lfloor L\rfloor + 1) = L - \lfloor L\rfloor,
      \qquad E[n_h] = L
    \]
    Same vertically, independently.
    \column{0.48\textwidth}
    \small
    \textbf{Pedestrian @ 250\,m} ($0.5 \times 1.8$\,m)\\[0.3em]
    \begin{tabular}{lcc}
      \toprule
      & horizontal & vertical \\
      \midrule
      spacing $s$ & 0.436\,m & 0.218\,m \\
      $L$ & 1.15 & 8.25 \\
      counts & 1 or 2 & 8 or 9 \\
      probability & 85\,\% / 15\,\% & 75\,\% / 25\,\% \\
      \bottomrule
    \end{tabular}\\[0.8em]
    $\Rightarrow$ mostly \textbf{a single column} of 8--9 beams:\\
    the horizontal resolution is the bottleneck
  \end{columns}
\end{frame}

\begin{frame}{Detection probability per beam}
  \begin{columns}[c]
    \column{0.52\textwidth}
    \small
    \textbf{Signal} (lidar equation, Lambertian, target $>$ spot):
    $P_r \propto \dfrac{\rho}{R^2}\,e^{-2\alpha R}$\\[0.5em]
    \textbf{Detection:} signal events $\sim$ Poisson($\lambda$), $\lambda \propto P_r$;
    return if $\ge 1$ event:
    \[
      p = 1 - e^{-\lambda} = 1 - \exp\!\Big(-C\,\frac{\rho}{R^2}\,e^{-2\alpha R}\Big)
    \]
    \textbf{Calibration} $p(210\,\text{m}, 10\,\%) = 0.9$:
    \[
      C = -\ln(1-P_0)\,\frac{R_0^2}{\rho_0}
    \]
    At 250\,m: $\ln 10 \cdot (210/250)^2 = 1.625 \Rightarrow p = 0.803$
    \column{0.44\textwidth}
    \small
    \centering
    \begin{tabular}{rcc}
      \toprule
      $R$ [m] & $p$ (10\,\%) & $p$ (5\,\%) \\
      \midrule
      150 & 0.989 & 0.895 \\
      200 & 0.921 & 0.719 \\
      \textbf{210} & \textbf{0.900} & 0.684 \\
      235 & 0.841 & 0.601 \\
      \textbf{250} & \textbf{0.803} & 0.556 \\
      275 & 0.739 & 0.489 \\
      300 & 0.676 & 0.431 \\
      \bottomrule
    \end{tabular}\\[0.5em]
    \scriptsize one free constant, fixed by the one datasheet point
  \end{columns}
\end{frame}

\begin{frame}{Statistics: worked example (pedestrian, 250\,m, 10\,\%, clear)}
  \small
  \[
    P(K \ge 5 \mid n) = 1 - \sum_{i=0}^{4} \binom{n}{i} p^i (1-p)^{n-i},
    \qquad
    P(K \ge 5) = \sum_{ij} w_{ij}\, P(K\ge5 \mid n_{ij})
  \]
  \centering
  \begin{tabular}{cccc}
    \toprule
    $n$ & weight $w$ & $P(K\ge5\mid n)$ & contribution \\
    \midrule
    $1\times8 = 8$ & $0.85\cdot0.75 = 0.64$ & 0.947 & 0.604 \\
    $1\times9 = 9$ & $0.85\cdot0.25 = 0.21$ & 0.982 & 0.209 \\
    $2\times8 = 16$ & $0.15\cdot0.75 = 0.11$ & 1.000 & 0.112 \\
    $2\times9 = 18$ & $0.15\cdot0.25 = 0.04$ & 1.000 & 0.037 \\
    \midrule
    & & & \fail{0.962} $< 0.99$ \\
    \bottomrule
  \end{tabular}\\[0.8em]
  $E[K] = 1.15 \cdot 8.25 \cdot 0.803 = 7.6 \ge 5$ \quad but \quad $P(K < 5) = 3.8\,\%$\\[0.3em]
  \textbf{the average passes, the probability fails}
\end{frame}

\begin{frame}{Verification}
  \small
  \begin{enumerate}
    \item \textbf{Calibration:} $p(210\,\text{m}, 10\,\%) = 0.900$ exactly
    \item \textbf{Hand calculation} (B5) = code output: 0.9618
    \item \textbf{Monte Carlo}, independent of the formulas:\\
      200\,000 frames: random grid offset $u \sim U[0,s)$, count beams, draw $K \sim \mathrm{Bin}(n,p)$
  \end{enumerate}
  \vspace{0.5em}
  \centering
  \begin{tabular}{lcc}
    \toprule
    pedestrian, 250\,m & formula & Monte Carlo \\
    \midrule
    $P(K \ge 5)$ & 96.18\,\% & 96.19\,\% \\
    $E[K]$ & 7.60 & 7.60 \\
    \bottomrule
  \end{tabular}\\[1em]
  \raggedright
  \textbf{Why not Poisson for $K$?} With $p = 0.8$ the binomial is much narrower:
  Poisson gives 87\,\% (too pessimistic)
\end{frame}
